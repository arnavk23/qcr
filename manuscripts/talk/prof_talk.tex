% ################################################################ PART I
\partslide{I}{The method we start from}{Replace short windows of a circuit with shorter equivalents from a precomputed database.}

\begin{frame}{Why shorten circuits at all?}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{itemize}\setlength{\itemsep}{0.6em}
        \item On today's hardware every gate costs fidelity. Roughly,
              \[\Pr[\text{success}] \approx \prod_{\text{gates } g} \bigl(1-e(g)\bigr),\]
              the \textbf{ESP}. Two-qubit errors are 10--100\(\times\) single-qubit ones.
        \item Compiling to a device's \textbf{native gates} leaves redundancy.
        \item Goal: an \emph{equivalent} circuit with fewer, or cheaper, native gates.
      \end{itemize}
    \end{column}
    \begin{column}{0.51\textwidth}
      \two{
      \centering
      \includegraphics[width=\linewidth]{figs/Motivation1.png}\\[0.3em]
      {\scriptsize\color{muted}H + CNOT compiled to ion-trap gates (8 gates) and the shortest equivalent (4 gates).}}
    \end{column}
  \end{columns}
  \src{Figure from arXiv:2502.14715, Fig.~1}
\end{frame}
\note{100 gates at 99 percent each give 37 percent success, 50 gates give 61 percent. That is the whole motivation.}

\begin{frame}{The method: a database and a sliding window}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \centering\includegraphics[width=0.7\linewidth]{figs/FullComputeGraphs2.png}\\[0.2em]
      {\scriptsize\color{muted}\textbf{Compute graph.} Nodes are circuits (as matrices), edges add one gate. Everything within
       radius \(r\) of the identity is stored with its shortest form. The paper uses \(r=5\)--\(6\).}
    \end{column}
    \begin{column}{0.49\textwidth}
      \two{
      \centering\includegraphics[width=0.66\linewidth]{figs/CodeReduceTokens.png}\\[0.2em]
      {\scriptsize\color{muted}\textbf{Window.} Take 2--8 gates on \(\le3\) wires, look them up, substitute a shorter word,
       shuffle commuting gates, repeat.}}
    \end{column}
  \end{columns}
  \two{\takeaway{Two knobs decide everything later: the window length \(w\) and the database radius \(r\).}}
  \src{Figures from arXiv:2502.14715, Figs.~2 and 3}
\end{frame}
\note{V1 searches randomly, V2 looks up the database, V3 adds a random forest that predicts reducible windows. The
database grows exponentially with r, which is why the paper keeps it small.}

% ################################################################ PART II
\partslide{II}{Exact ground truth}{Every gate the paper uses is Clifford, so the search space is a finite group we can enumerate completely.}

\begin{frame}{Every gate in the paper is Clifford}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{itemize}\setlength{\itemsep}{0.5em}
        \item \(\pm\tfrac{\pi}{2}\) rotations, \(R_{XX}(\pm\tfrac{\pi}{2})\) and CZ map Paulis to Paulis.
        \item So every circuit is an element of the finite \textbf{Clifford group}: 24 elements for 1 qubit,
              11,520 for 2, \textbf{92,897,280} for 3.
        \item We enumerated all of them for 3 qubits and stored the exact shortest length of each (25\,min, 6\,GB).
      \end{itemize}
      \two{\takeaway{Optimality is no longer vague. Each test circuit has a proven shortest length.}}
    \end{column}
    \begin{column}{0.50\textwidth}
      \centering\includegraphics[width=\linewidth]{figs/ext_growth.png}\\
      {\scriptsize\color{muted}Elements by optimal length (3 qubits). The paper's depth-5 database (shaded) covers 0.15\,\% of the group.}
    \end{column}
  \end{columns}
\end{frame}
\note{The compute graph in the paper is just a ball in the Cayley graph of this group. The optimiser we test later never
uses this table, it is only the referee. Beyond Clifford we computed exact 2-qubit Clifford+T distances.}

\begin{frame}{How far from optimal is the paper's method?}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \centering\footnotesize\renewcommand{\arraystretch}{1.25}
      \small 3 qubits, 300 random gates, 20 circuits per row\\[0.4em]
      \begin{tabular}{@{}lccc@{}}
        \toprule
        & \textbf{optimum} & paper's method & Qiskit L3\\\midrule
        ion trap  & \textbf{10.4} & 47.1 & 127.2\\
        IBM-style & \textbf{12.2} & 36.8 & 104.0\\\bottomrule
      \end{tabular}
      \vspace{0.8em}
      \two{
      \begin{alertblock}{The gap}
        The method's output is \textbf{4.5\(\times\)} (ion) and \textbf{3.0\(\times\)} (IBM-style) the optimum, on its own benchmark.
        Our port matches its published number (47.1 vs.\ 45), so this is the method, not our code.
      \end{alertblock}}
    \end{column}
    \begin{column}{0.52\textwidth}
      \centering\includegraphics[width=\linewidth]{figs/fig_example_paper.pdf}\\
      {\scriptsize\color{muted}A typical circuit: 80 gates in, the paper's method stops at 24, the proven optimum is 11.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Why it falls short: the database is too small}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \centering\includegraphics[width=0.88\linewidth]{figs/ext_window.png}\\
      {\scriptsize\color{muted}With exact lookups, a window of \(w\ge12\) (ion) or \(16\) (IBM-style) reaches the proven optimum on all 40 circuits.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \two{
      \centering\includegraphics[width=0.62\linewidth]{figs/ext_grid.png}\\
      {\scriptsize\color{muted}Mean excess over the optimum by window \(w\) and radius \(r\). The paper's \((8,5)\) in red.}
      \begin{itemize}\setlength{\itemsep}{0.2em}\small
        \item Larger window at \(r=5\): excess 46 \(\to\) 43.
        \item Larger database at \(w=8\): 46 \(\to\) 18.
        \item \(w\ge12\) and \(r\ge8\): excess \(\le2\).
      \end{itemize}}
    \end{column}
  \end{columns}
  \two{\takeaway{The database radius is the binding constraint. Our theorem predicts that both knobs must be large enough.}}
\end{frame}
\note{Why so sharp: once the window is longer than the group's diameter region, one more sweep is exact (the cascade
theorem). At 4 qubits the group has 1.2e13 elements, so we prove a lower bound instead; the remaining gap is at most
about 4 times.}

% ################################################################ PART III
\partslide{III}{A general optimiser}{Drop the Clifford crutch and add continuous angles, connectivity, calibrated noise, and exactness including phase.}

\begin{frame}{The engine}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{tikzpicture}[node distance=0.32cm, every node/.style={font=\footnotesize},
          box/.style={draw=ibar,fill=ifoot,rounded corners=2pt,minimum width=4.9cm,minimum height=0.62cm,align=center},
          arr/.style={-{Stealth[length=2mm]},ibar,thick}]
        \node[box] (w) {\textbf{convex window} on \(\le3\) coupled wires\\{\scriptsize grown by exact commutation}};
        \node[box,below=of w] (d) {\textbf{database lookup} (canonical hash)};
        \node[box,below=of d] (m) {\textbf{meet in the middle}, \(U=R\,A\), \(A\in B(2)\)};
        \node[box,below=of m] (r) {\textbf{analytic resynthesis} on \(\le2\) wires\\{\scriptsize KAK (min.\ CZ) / Weyl \(\to\) MS\((\theta)\) / Euler}};
        \node[box,below=of r,fill=igreen!12,draw=igreen] (a) {keep the \textbf{cheapest} legal replacement};
        \draw[arr] (w)--(d); \draw[arr] (d)--(m); \draw[arr] (m)--(r); \draw[arr] (r)--(a);
        \draw[arr] (a.east) -- ++(0.35,0) |- node[pos=0.25,right,align=left,font=\scriptsize\color{muted}]{sweep +\\ random\\ commuting\\ shuffles} (w.east);
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.46\textwidth}
      \two{
      \begin{itemize}\setlength{\itemsep}{0.45em}\small
        \item \textbf{Cost} \(c_{\rm cal}(g)=-\log(1-e(g))\) from calibration, so minimising cost maximises ESP.
        \item \textbf{Exact including global phase}: phases are tracked and repaired once at the end.
        \item \textbf{Never worse} than its input, by construction.
        \item \textbf{Any gate set}: databases of 0.3--31\,M classes for IBM Heron and trapped ions.
        \item Runs \emph{after} any compiler, as a post-pass.
      \end{itemize}}
    \end{column}
  \end{columns}
\end{frame}
\note{Convex windows can reach past gates on other qubits that commute out of the way, which is what made the 4-qubit
results jump. Approximate and state-aware modes also exist; headline numbers use exact mode.}

\begin{frame}{Every output is verified, even beyond 25 qubits}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{itemize}\setlength{\itemsep}{0.45em}\small
        \item \textbf{Up to 24 qubits.} \(\lVert (U-V)\psi\rVert\) on random states: 250/250 paper outputs, 720/720 suite
              outputs, 57 large outputs (max \(1.4\cdot10^{-10}\)).
        \item \textbf{Clifford, Clifford+T.} Exact arithmetic.
        \item \textbf{Beyond.} Statevector is out of reach, and MQT QCEC gave no answer on routed 23-qubit adders.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \two{
      \textbf{Replay certificates}
      \begin{itemize}\setlength{\itemsep}{0.3em}\small
        \item The optimiser logs every step. An independent checker replays it with small matrices on a few wires.
        \item \textbf{171/176} large outputs certified, up to 544,922 steps, at most 6.8\,s each.
        \item 5 more were rejected by an older checker limit and are being re-checked.
      \end{itemize}}
    \end{column}
  \end{columns}
  \two{\takeaway{Checking a whole circuit becomes many tiny local checks, so cost grows with the steps, not the width.}}
\end{frame}

\partslide{V}{Wrap-up}{Where it stands and what comes next.}

\begin{frame}{Limitations and next steps}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Limitations}
      \begin{itemize}\setlength{\itemsep}{0.35em}\small
        \item Noise models, no real-hardware runs yet.
        \item Windows of at most 3 qubits, single-threaded Python, hours at 100 qubits.
        \item Baselines are Qiskit, TKET and BQSKit at default settings and one seed.
      \end{itemize}
    \end{column}

\end{frame}

\begin{frame}{Summary}
  \begin{enumerate}\setlength{\itemsep}{0.8em}
    \item The original method is \textbf{3--4.5\(\times\)} off the proven optimum, and the database radius is why.
    \item A general optimiser with deep databases and convex windows reaches the optimum on \textbf{35/40} circuits
          without knowing the group.
    \item On real algorithms it is a safe post-pass: after Qiskit L3 it never adds a 2q gate and removes 13--19\,\% of
          gates, and after BQSKit it removes 16\,\% more.
    \item Every output is verified, by simulation or by replay certificate.
  \end{enumerate}
\end{frame}
