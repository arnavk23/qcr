\partslide{I}{The method we start from}{Replace short windows of a circuit with shorter equivalents from a precomputed database.}

\begin{frame}{Why shorten circuits at all?}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \small
      On noisy hardware every gate can fail. A standard estimate of the chance that the whole circuit runs correctly is
      \[\text{ESP}=\prod_{\text{gates } g} \bigl(1-e(g)\bigr).\]
      \centering\scriptsize\renewcommand{\arraystretch}{1.15}
      \textbf{Error rates per gate (medians)}\\[0.2em]
      \begin{tabular}{@{}lccc@{}}
        \toprule
        & 1q gate & 2q gate & 2q / 1q\\\midrule
        IBM Heron\(^\ast\) & \(2.7\cdot10^{-4}\) & \(4.1\cdot10^{-3}\) & \textbf{15\(\times\)}\\
        trapped ion\(^\dagger\) & \(3\cdot10^{-5}\) & \(10^{-3}\) & \textbf{33\(\times\)}\\\bottomrule
      \end{tabular}\\[0.1em]
      {\tiny\color{muted}\(^\ast\)FakeTorino calibration, 133 qubits, 278 couplers. \(^\dagger\)model rates used in the paper.}\\[0.4em]
      \textbf{Fewer gates, higher ESP}\\[0.2em]
      \begin{tabular}{@{}lc@{}}
        \toprule
        gates, each 99\,\% & ESP\\\midrule
        100 & \(0.99^{100}=37\,\%\)\\
        50 & \(0.99^{50}=61\,\%\)\\\bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.49\textwidth}
      \two{
      \centering
      \includegraphics[width=\linewidth]{figs/Motivation1.png}\\[0.3em]
      {\scriptsize\color{muted}H + CNOT compiled to ion-trap gates (8 gates) and the shortest equivalent (4 gates).}}
    \end{column}
  \end{columns}
  \src{Error rates: \texttt{experiments/make\_talk\_figs.py errors} (Qiskit FakeTorino calibration) and the ion model in the paper, Sec.~3. Figure: arXiv:2502.14715, Fig.~1}
\end{frame}
\note{Two-qubit gates are the expensive ones, 15 times worse than one-qubit gates on IBM Heron. Halving the gate count
takes success from 37 to 61 percent.}

\begin{frame}{The method: a database and a sliding window}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \centering\includegraphics[width=0.7\linewidth]{figs/FullComputeGraphs2.png}\\[0.2em]
      {\scriptsize\color{muted}\textbf{Compute graph.} Nodes are circuits (as matrices), edges add one gate. Everything within
       radius \(r\) of the identity is stored with its shortest form. The paper uses \(r=5\)--\(6\).}
    \end{column}
    \begin{column}{0.49\textwidth}
      \two{
      \centering\includegraphics[width=0.66\linewidth]{figs/CodeReduceTokens.png}\\[0.2em]
      {\scriptsize\color{muted}\textbf{Window.} Take 2--8 gates on \(\le3\) wires, look them up, substitute a shorter word,
       shuffle commuting gates, repeat.}}
    \end{column}
  \end{columns}
  \two{\takeaway{Two knobs decide everything later: the window length \(w\) and the database radius \(r\).}}
  \src{Figures from arXiv:2502.14715, Figs.~2 and 3}
\end{frame}
\note{V1 searches randomly, V2 looks up the database, V3 adds a random forest that predicts reducible windows. The
database grows exponentially with r, which is why the paper keeps it small.}

\partslide{II}{Exact ground truth}{The gates behind the paper's benchmarks are Clifford, so the search space is a finite group we can enumerate completely.}

\begin{frame}{Where the Clifford claim comes from: their Table I}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \centering\includegraphics[width=\linewidth]{figs/src_tableI.png}\\
      {\scriptsize\color{muted}arXiv:2502.14715, p.~4, Table I (cropped, unchanged)}
    \end{column}
    \begin{column}{0.54\textwidth}\small
      \two{
      \textbf{Only \(\pm\tfrac{\pi}{2}\) angles give these operator counts}\\[0.2em]
      {\scriptsize\renewcommand{\arraystretch}{1.15}
      \begin{tabular}{@{}lcc@{}}
        \toprule
        & 2 qubits & 3 qubits\\\midrule
        \(R_X,R_Y,R_Z(\pm\tfrac\pi2)\) on each qubit & \(3\cdot2\cdot2=12\) & \(3\cdot2\cdot3=18\)\\
        \(R_{XX}(\pm\tfrac\pi2)\) on each pair & 2 & 6\\\midrule

      \end{tabular}}\\[0.5em]

      The authors' demo files also contain only \(\pm\tfrac\pi2\) angles.\\[0.5em]
      \(\pm\tfrac\pi2\) rotations and \(R_{XX}(\tfrac\pi2)\) map Pauli operators to Pauli operators, which is the
      definition of a \textbf{Clifford} gate.}

\end{frame}
\note{The arXiv version never prints the angles. The operator counts pin them down. The journal version adds
pi/4 for its later experiments, which is no longer Clifford. That case is Clifford+T, and we handle it separately
with exact arithmetic (Result 1).}

\begin{frame}[fragile]{How the code checks it}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \footnotesize\textbf{Every gate is tested before the search starts}\\[0.2em]
\begin{semiverbatim}\scriptsize
\textcolor{muted}{# qcr/tableau.py, Tableau.row_table}
for b in range(4**n):            \textcolor{muted}{# every Pauli P}
    Q = U @ P[b] @ U^\dag           \textcolor{muted}{# conjugate it}
    for c in range(4**n):
        ov = <P[c], Q> / 2**n
        if abs(abs(ov) - 1) < 1e-8: \textcolor{muted}{# Q = +-P[c]}
            ...; break
    else:
        raise ValueError("gate is not Clifford")
\end{semiverbatim}
      \vspace{0.3em}
      \textbf{Then the whole group is enumerated}\\[0.2em]
\begin{semiverbatim}\scriptsize
\textcolor{muted}{# results/cayley3_ion_pm.log (last lines)}
n=3 depth 15: +37,888 (total 92,891,648) 1489s
n=3 depth 16: +5,120  (total 92,896,768) 1491s
n=3 depth 17: +512    (total 92,897,280) 1492s
\end{semiverbatim}
    \end{column}
    \begin{column}{0.43\textwidth}
      \two{
      \centering\footnotesize\renewcommand{\arraystretch}{1.2}
      \textbf{Group size: enumeration vs.\ formula}\\[0.2em]
      \begin{tabular}{@{}cccc@{}}
        \toprule
        \(n\) & found & \(2^{n^2+2n}\prod_j(4^j-1)\) & test\\\midrule
        1 & 24 & 24 & \good{pass}\\
        2 & 11,520 & 11,520 & \good{pass}\\
        3 & 92,897,280 & 92,897,280 & \good{pass}\\\bottomrule
      \end{tabular}\\[0.6em]
      \begin{tabular}{@{}lcc@{}}
        \toprule
        3 qubits & ion & IBM-style\\\midrule
        diameter (longest optimum) & 17 & 19\\
        mean optimal length & 10.72 & 12.34\\
        enumeration time & 25 min & 27 min\\\bottomrule
      \end{tabular}\\[0.3em]
      {\scriptsize\color{muted}\texttt{experiments/ext\_cayley3.py}, \texttt{results/cayley3\_\{ion\_pm,nisq\}.json}, \texttt{test/test\_clifford.py}}}
    \end{column}
  \end{columns}
\end{frame}
\note{A Clifford circuit is stored as its tableau, a 64-bit number for 3 qubits. Breadth-first search from the identity
visits each element once, at its exact shortest length. Stopping exactly at the textbook group order is the check that
nothing was missed.}

\partslide{III}{A general optimiser}{Drop the Clifford crutch and add continuous angles, connectivity, calibrated noise, and exactness including phase.}

\begin{frame}{The engine}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{tikzpicture}[node distance=0.32cm, every node/.style={font=\footnotesize},
          box/.style={draw=ibar,fill=ifoot,rounded corners=2pt,minimum width=4.9cm,minimum height=0.62cm,align=center},
          arr/.style={-{Stealth[length=2mm]},ibar,thick}]
        \node[box] (w) {\textbf{convex window} on \(\le3\) coupled wires\\{\scriptsize grown by exact commutation}};
        \node[box,below=of w] (d) {\textbf{database lookup} (canonical hash)};
        \node[box,below=of d] (m) {\textbf{meet in the middle}, \(U=R\,A\), \(A\in B(2)\)};
        \node[box,below=of m] (r) {\textbf{analytic resynthesis} on \(\le2\) wires\\{\scriptsize KAK (min.\ CZ) / Weyl \(\to\) MS\((\theta)\) / Euler}};
        \node[box,below=of r,fill=igreen!12,draw=igreen] (a) {keep the \textbf{cheapest} legal replacement};
        \draw[arr] (w)--(d); \draw[arr] (d)--(m); \draw[arr] (m)--(r); \draw[arr] (r)--(a);
        \draw[arr] (a.east) -- ++(0.35,0) |- node[pos=0.25,right,align=left,font=\scriptsize\color{muted}]{sweep +\\ random\\ commuting\\ shuffles} (w.east);
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.46\textwidth}
      \two{
      \begin{itemize}\setlength{\itemsep}{0.45em}\small
        \item \textbf{Cost} \(c_{\rm cal}(g)=-\log(1-e(g))\) from calibration, so minimising cost maximises ESP.
        \item \textbf{Exact including global phase}: phases are tracked and repaired once at the end.
        \item \textbf{Never worse} than its input, by construction.
        \item \textbf{Any gate set}, through databases built per device (next slide).
        \item Runs \emph{after} any compiler, as a post-pass.
      \end{itemize}}
    \end{column}
  \end{columns}
\end{frame}
\note{Convex windows can reach past gates on other qubits that commute out of the way, which is what made the 4-qubit
results jump. Approximate and state-aware modes also exist. Headline numbers use exact mode.}

\begin{frame}{Limitations and next steps}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Limitations}
      \begin{itemize}\setlength{\itemsep}{0.35em}\small
        \item Real hardware so far shows no measurable gain (15 small circuits, Result 6).
        \item Windows of at most 3 qubits, single-threaded Python, hours at 100 qubits.
        \item VOQC beats us alone on the Nam set, BQSKit on 2q gates.
        \item Quartz not run, QUESO at 5 min, BQSKit on the smaller large circuits only.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \two{
      \textbf{Next}
      \begin{itemize}\setlength{\itemsep}{0.35em}\small
        \item Hardware runs with more repeats, on circuits where we remove 2q gates.
        \item Reach VOQC's rotation merging and BQSKit's 2q counts exactly (phase-polynomial passes, larger windows).
        \item Quartz baseline, QUESO with its full hour.
      \end{itemize}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Summary}
  \begin{enumerate}\setlength{\itemsep}{0.6em}
    \item The paper's benchmark gates are Clifford, so the optimum is computable. The method is \textbf{3--4.5\(\times\)}
          off it, and the database radius is why.
    \item A general optimiser with deep databases and convex windows reaches the optimum on \textbf{35/40} circuits
          without knowing the group.
    \item On real algorithms it is a safe post-pass. After Qiskit L3 it never adds a 2q gate and removes 13--19\,\% of
          gates, and after BQSKit it removes 16\,\% more.
    \item The gains are stable over seeds, and window length and database depth drive them.
    \item Against VOQC and QUESO, ours is second alone and VOQC followed by ours is best.
    \item Every output is verified. On real hardware the effect is not yet measurable.
  \end{enumerate}
\end{frame}
