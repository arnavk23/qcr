\partslide{I}{The paper}{Rosenhahn, Osborne \& Hirche shorten circuits by replacing short windows with shorter equivalents from a precomputed database.}

\begin{frame}{Why shorten circuits at all?}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{itemize}\setlength{\itemsep}{0.6em}
        \item On NISQ hardware every gate costs fidelity. Roughly,
              \[\Pr[\text{success}] \approx \prod_{\text{gates } g} \bigl(1-e(g)\bigr),\]
              with two-qubit errors \(\sim10^{-3}\)--\(10^{-2}\), 10--100\(\times\) single-qubit ones.
        \item Compiling to a device's \textbf{native alphabet} leaves redundancy.
              One textbook gate becomes several native ones, and neighbours stop cancelling.
        \item The goal is an equivalent circuit with fewer (or cheaper) native gates.
      \end{itemize}
    \end{column}
    \begin{column}{0.51\textwidth}
      \two{
      \centering
      \includegraphics[width=\linewidth]{figs/Motivation1.png}\\[0.3em]
      {\scriptsize\color{muted}The paper's motivating example, H + CNOT compiled to the ion-trap alphabet (8 gates)
       and the shortest equivalent native word (4 gates). We confirmed by exhaustive search that no 3-gate version exists.}}
    \end{column}
  \end{columns}
  \src{Figure from arXiv:2502.14715, Fig.~1}
\end{frame}

\begin{frame}{Step 1. The compute graph, a database of shortest circuits}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{itemize}\setlength{\itemsep}{0.5em}
        \item Fix a \textbf{discrete alphabet} on 2--3 qubits
          \begin{itemize}
            \item ion trap, \(R_{x},R_y,R_z(\pm\tfrac{\pi}{2})\), \(R_{XX}(\pm\tfrac{\pi}{2})\)
            \item IBM-style, \(R_x,R_z(\pm\tfrac{\pi}{2})\), CZ
          \end{itemize}
        \item Breadth-first search over words. Each node is a unitary (up to global phase), stored with its
              \emph{shortest} word.
        \item Depth \(d\) = maximal word length enumerated.
      \end{itemize}
      \two{
      \vspace{0.4em}\centering\footnotesize
      \begin{tabular}{@{}lrrrrr@{}}
        \toprule
        depth & 1 & 2 & 3 & 4 & 5\\\midrule
        2q, 14 ops & 15 & 114 & 584 & 2,024 & 4,512\\
        3q, 24 ops & 25 & 337 & 3,215 & 23,622 & 137,572\\\bottomrule
      \end{tabular}\\[0.2em]
      {\scriptsize\color{muted}Table I node counts. Our rebuild matches the paper exactly.}}
    \end{column}
    \begin{column}{0.51\textwidth}
      \centering\includegraphics[width=0.9\linewidth]{figs/FullComputeGraphs2.png}\\[0.2em]
      {\scriptsize\color{muted}Compute graphs of growing depth (paper, Fig.~2).}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Step 2. Slide a window, replace, repeat}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \centering\includegraphics[width=0.85\linewidth]{figs/CodeReduceTokens.png}\\[0.2em]
      {\scriptsize\color{muted}Circuit as a word of tokens, with windows replaced by shorter database words (paper, Fig.~3).}
      \vspace{0.5em}
      \begin{itemize}\setlength{\itemsep}{0.3em}\small
        \item Pick a window of 2--8 gates on \(\le 3\) wires.
        \item Look its unitary up and substitute a shorter word if one exists.
        \item Shuffle commuting gates to expose new windows, and repeat until no progress.
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      \two{
      \centering\includegraphics[width=\linewidth]{figs/optVariantsW.png}\\[0.3em]

        \item \textbf{V1.} Random sampling for a shorter equivalent word.
        \item \textbf{V2.} Database lookup.
        \item \textbf{V3.} A random forest first predicts whether a window is reducible, and only then is it looked up.
      \end{itemize}}
    \end{column}
  \end{columns}

\end{frame}

\partslide{III}{Exact ground truth}{The observation the paper misses. Every gate it uses is Clifford, so the search space is a finite group we can enumerate completely.}

\begin{frame}{Key observation, every gate in the paper is Clifford}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{itemize}\setlength{\itemsep}{0.5em}
        \item \(\pm\tfrac{\pi}{2}\) rotations, \(R_{XX}(\pm\tfrac{\pi}{2})\) and CZ map Paulis to Paulis.
        \item So every circuit is an element of the finite \textbf{Clifford group} \(\mathcal C_n\) (mod phase), of order
              \[2^{n^2+2n}\prod_{j=1}^n (4^j-1)\]
              (24 for \(n=1\), 11,520 for \(n=2\), \textbf{92,897,280} for \(n=3\)).
        \item The paper's ``compute graph'' is a ball in the \textbf{Cayley graph} of this group.
      \end{itemize}
    \end{column}
    \begin{column}{0.50\textwidth}
      \centering\includegraphics[width=0.78\linewidth]{figs/notes_1q_graph.png}\\
      {\scriptsize\color{muted}For one qubit the complete compute graph has exactly 24 nodes and diameter 3.}
    \end{column}
  \end{columns}
  \two{\takeaway{Optimality is no longer a vague question. Each circuit has an exact shortest length, computable by search over one finite graph.}}
\end{frame}

\begin{frame}{Enumerating all 92.9 million 3-qubit Cliffords}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \begin{itemize}\setlength{\itemsep}{0.5em}
        \item A Clifford mod phase is its signed \textbf{tableau}, which for \(n\le3\) packs into one 64-bit word.
        \item Gates act row-wise through lookup tables, so each BFS level is a vectorised gather. Exact membership, no tolerance.
        \item Full BFS takes 25\,min and \(\approx\)6\,GB per alphabet.
      \end{itemize}
      \vspace{0.4em}
      \two{
      \centering\footnotesize\renewcommand{\arraystretch}{1.15}
      \begin{tabular}{@{}lrrr@{}}
        \toprule
        alphabet & \(n\) & diameter & mean optimal length\\\midrule
        ion trap & 2 & 10 & 5.89\\
        IBM-style & 2 & 11 & 6.92\\
        ion trap & 3 & \textbf{17} & \textbf{10.72}\\

\partslide{IV}{Theory}{Local rewriting as rewriting on a Cayley graph. When is it exact, what can it never do, and what do random benchmarks measure?}

\begin{frame}{The model}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{itemize}\setlength{\itemsep}{0.5em}
        \item Alphabet \(\Gamma\), group \(G=\langle\Gamma\rangle\), word length \(|g|\), ball \(B(r)\), diameter \(D\).
        \item Compute graph of depth \(d\) \(=\) ball \(B(d)\) in \(\mathrm{Cay}(G,\Gamma)\), and stored words are geodesics.
        \item The rewriting system \(\mathcal R(w,r)\) replaces a factor of length \(\le w\) by an equivalent geodesic of length \(\le r\).
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \two{
      \begin{itemize}\setlength{\itemsep}{0.5em}
        \item Swapping commuting neighbours = moves in the \textbf{trace monoid}.
        \item Factors of trace-equivalent words = \textbf{convex} sets of the circuit DAG.
        \item So the paper's random commuting shuffles are a \emph{stochastic enumeration of convex windows}, and we enumerate them directly.
        \item The paper's V2 operates at \((w,r)=(8,5)\).
      \end{itemize}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Theorem 1. Local = global above the diameter}
  \begin{block}{Exactness threshold}
    Let \(G\) be finite with diameter \(D\). If \(w\ge D+1\) and \(r\ge D\), every \((w,r)\)-reduced word is a geodesic.
  \end{block}
  \small\textit{Proof idea.} A reduced word longer than \(D\) would have a first factor of length \(D+1\), which is not geodesic and fits in the window. So the word has length \(\le D<w\), and then the whole word is one window.
  \vspace{0.6em}
  \two{
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Cascade}
        If at any time the current length \(\ell \le w\), the next sweep returns a geodesic. The relevant window is
        \(w^\star=\min\{w:\ell(w)\le w\}\).
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}\setlength{\itemsep}{0.3em}\small
        \item Measured \(w^\star\approx12\) (ion) / 16 (IBM-style) for 3 qubits, well below \(D+1=18\).
        \item The paper's \((8,5)\) is below both thresholds.
        \item For continuous gate sets \(D=\infty\) and no finite window is exact. Hence ground truth must come from the discrete case.
      \end{itemize}
    \end{column}
  \end{columns}}
\end{frame}

\begin{frame}{Theorem 2. Gate-count parity}
  \begin{block}{Parity}
    (a) For \(n\le2\) the Cayley graph is \textbf{bipartite}, so every equivalent rewrite removes an \emph{even} number of gates.\\
    (b) For \(n\ge3\), with any generating set containing square roots of each \(X_i\), \(Z_i\), it is \textbf{never} bipartite.
  \end{block}
  \small\textit{Proof idea.} Bipartite \(\iff\) a sign character \(\chi\) with \(\chi(\Gamma)=-1\). For \(n\ge3\), \(\chi\) kills the Paulis and factors through \(\mathrm{Sp}(2n,2)\), which is simple, so \(\chi\equiv1\).
  \vspace{0.6em}
  \two{
  \begin{itemize}\setlength{\itemsep}{0.4em}
    \item Exhaustive search at 3 qubits finds a shortest odd relation of length \(\rho=11\) (ion) / 9 (IBM-style).
          Changing parity needs a window of \(\ge6\) gates, and smaller windows can never reach an optimum of the other parity.
    \item The paper's Fig.~4 shows a \(6\to3\) replacement on two qubits, which is \textbf{impossible} with the stated gate set.
  \end{itemize}}
\end{frame}

\begin{frame}{Theorem 3. Random-circuit benchmarks measure a constant}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Benchmarks mix}
        A random word of length \(L\) is within \(\tfrac12\sqrt{|G|}\,\beta^L\) (total variation) of a uniform group element.
        So \(\mathbb E[\text{optimal length}]\to\) the mean distance in \(G\).
      \end{block}
      \vspace{0.3em}
      \small For 2 qubits (ion trap) \(\beta=0.937\) (sparse diagonalisation).
    \end{column}
    \begin{column}{0.50\textwidth}
      \centering\includegraphics[width=\linewidth]{figs/ext_mixing.png}
    \end{column}
  \end{columns}
  \two{\takeaway{``Reduced from 300 to 45 gates'' mostly measures the input length. Beyond \(L\approx100\), every random 3-qubit ion circuit has optimum \(\approx10.7\) on average, whatever \(L\) is.}}
\end{frame}

\begin{frame}{Lower bounds for every method (4 qubits)}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Counting bound}
        \(\Pr[|g|\le d]\le|B(d)|/|G|\), and spheres are submultiplicative, \(|S(a+b)|\le|S(a)||S(b)|\).
        Exact spheres up to a small depth bound \(\mathbb E|g|\) from below for any optimiser.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \two{
      \begin{itemize}\setlength{\itemsep}{0.5em}
        \item Check at \(n=3\), bound 7.9 vs.\ truth 10.72.
        \item \(n=4\) (\(1.2\times10^{13}\) elements), spheres to depth 6 give \(\mathbb E|g|\ge\) \textbf{11.1} (ion) / \textbf{12.9} (IBM-style).
        \item The best achieved is 45.6 / 49.3. Closing that gap is an open problem the framework makes precise.
      \end{itemize}}
    \end{column}
  \end{columns}
\end{frame}

\partslide{V}{A general optimiser}{Drop the Clifford crutch and add continuous angles, connectivity, calibrated noise, and a precise notion of ``equivalent''.}

\begin{frame}{Why go beyond Clifford?}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{The Clifford results don't transfer}
      \begin{itemize}\setlength{\itemsep}{0.4em}
        \item Real compilers emit \textbf{arbitrary angles} (\(R_z(\theta)\), MS\((\theta)\)), so there is no finite group and no table.
        \item Real chips have \textbf{connectivity} (heavy-hex) and per-gate \textbf{calibrated errors}.
        \item ``Up to global phase'' is \textbf{unsafe} once a block is controlled.
        \item Total gate count is the wrong objective when 2q gates dominate error.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \two{
      \textbf{Design decisions}
      \begin{itemize}\setlength{\itemsep}{0.4em}
        \item \textbf{No} Clifford assumption in the method. Clifford tables are used only to \emph{evaluate}.
        \item Devices are IBM Heron (FakeTorino with \(R_z(\theta)\), \(\sqrt X\), \(X\), CZ, heavy-hex, calibrated) and a trapped-ion model
              (\(R_{x,y,z}(\theta)\), \(R_{XX}(\theta)\), all-to-all).
        \item Semantics are exact incl.\ phase, state/measurement-aware, or approximate with tracked error.
        \item The objective is chosen by noisy simulation, not assumed.
      \end{itemize}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{One idea, equivalence = group action and key = canonical orbit}
  \footnotesize\renewcommand{\arraystretch}{1.1}
  \begin{tabularx}{\textwidth}{@{}>{\raggedright\arraybackslash}p{0.30\textwidth}>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}p{0.28\textwidth}@{}}
    \toprule
    \textbf{notion} & \(U \sim V\) \textbf{iff} & \textbf{canonical database key}\\
    \midrule
    exact & \(V = U\) & raw matrix\\
    up to phase (the paper) & \(V = e^{i\varphi} U\) & phase-normalised matrix\\
    input-aware (at start) & \(V|0\rangle = e^{i\varphi}U|0\rangle\) & phase-normalised \(U|0\rangle\)\\
        output-aware (before readout) & \(V = D\,U\), \(D\) diagonal & rows phase-normalised\\
    symmetry & \(V = P U P^{\dagger}\), \(P\) a chip symmetry & minimum over the orbit\\
    approximate & \(\lVert U - V\rVert \le \varepsilon\) & exact key + verified distance\\
    \bottomrule
  \end{tabularx}
  \vspace{0.6em}
  \uncover<2->{%
  \begin{block}{One engine, many semantics}
    The database stores 64-bit hashes of canonical forms, so one lookup finds the whole class.
    New semantics = new key function. Every hit is re-verified numerically.
  \end{block}}
\end{frame}

\begin{frame}{Why the relaxed semantics are safe}
  \begin{block}{Proposition 1 (state- and measurement-aware rewriting)}
    The program prepares \(|0^n\rangle\), runs the circuit and measures all qubits. Let \(S\) be a convex window on wires \(W\).
    If (a) nothing precedes \(S\) on \(W\) and \(V|0_W\rangle = e^{i\varphi}U|0_W\rangle\), or
    (b) nothing follows \(S\) on \(W\) and \(V = DU\) with \(D\) diagonal,
    then replacing \(U\) by \(V\) leaves the output distribution unchanged, and so does any sequence of such rewrites.
  \end{block}
  \small\textit{Why.} In (a) the input is a product state, and in (b) \(|\langle x|D\psi\rangle|^2=|\langle x|\psi\rangle|^2\). The invariant is the output distribution.
  \vspace{0.5em}
  \two{
  \begin{block}{Proposition 2 (approximate rewriting)}
    If rewrite \(i\) has operator-norm error \(\varepsilon_i\), the final circuit is within \(\sum_i\varepsilon_i\) of the original (including phase in exact mode).
  \end{block}
  \small\textit{Why.} Telescoping, and \(\lVert A(X-Y)B\rVert=\lVert X-Y\rVert\) for unitary \(A,B\). The optimiser reports this bound with every circuit.}
\end{frame}

\begin{frame}{Global phase and cost, the details that matter}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Exact including global phase}
      \begin{itemize}\setlength{\itemsep}{0.4em}
        \item Inside windows, match up to phase and record each phase.
        \item Native gates reach only a \emph{discrete} set of phases, multiples of \(\pi/4\) (Heron) and \(\pi/2\) (ion). Snap, or reject.
        \item Repair once at the end with the cheapest gadget. On ions absorb \(\pm i\) into an existing MS gate,
              \(R_{XX}(\theta\pm\pi)\,R_x(\pi)^{\otimes 2} = \pm i\,R_{XX}(\theta)\).
        \item Never return a costlier circuit than the input.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \two{
      \textbf{Three candidate objectives}
      \[
        c_{\rm count}=1,\qquad c_{w2}=\begin{cases}10&\text{2q}\\1&\text{1q}\end{cases}
      \]
      \[
        c_{\rm cal}(g)=-\log\bigl(1-e(g)\bigr)
      \]
      \begin{itemize}\setlength{\itemsep}{0.4em}
        \item \(e^{-c_{\rm cal}}\) = estimated success probability (ESP), so minimising \(c_{\rm cal}\) maximises ESP.
        \item \(e(g)\) from FakeTorino calibration per qubit / pair, the ion model uses \(3\cdot10^{-5}\) / \(10^{-3}\), and \(R_z\) is virtual.
        \item Which one wins is decided by noisy simulation (Part VI).
      \end{itemize}}
    \end{column}
  \end{columns}
\end{frame}

\partslide{VII}{Wrap-up}{Lessons, limitations, critique of the original paper, and next steps.}

\begin{frame}{Lesson, ``exact'' is easy to get subtly wrong}
  Found on Nam \texttt{pf1\_10}/\texttt{pf2\_10} (rotations of \(\sim10^{-7}\)\,rad), whose outputs at first verified only to \(10^{-6}\).
  \vspace{0.6em}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{alertblock}{Pitfalls}
        \begin{itemize}
          \item \texttt{np.allclose} has a default \emph{relative} tolerance \(10^{-5}\), so near-commuting gates were swapped.
          \item \(|\mathrm{tr}(U^\dagger V)|/d \approx 1\) is \emph{quadratic} in the error and accepts \(\lVert U-V\rVert \sim 10^{-4}\).
          \item Native gates reach only a \emph{discrete} set of global phases.
        \end{itemize}
      \end{alertblock}
    \end{column}
    \begin{column}{0.49\textwidth}
      \two{
      \begin{exampleblock}{Fixes}
        \begin{itemize}
          \item Linear operator-distance tests, one tolerance \(10^{-10}\), \texttt{rtol=0}.
          \item Replacement phases snapped to the reachable set, with residuals rejected or charged to \(\varepsilon\).
          \item On ions \(\pm i\) is absorbed into an existing MS gate.
        \end{itemize}
      \end{exampleblock}}
    \end{column}
  \end{columns}
  \two{\takeaway{\texttt{pf1\_10} now verifies to \(5\cdot10^{-13}\) (was \(1.8\cdot10^{-7}\)). The original paper uses three different tolerances up to \(10^{-3}\).}}
\end{frame}

\begin{frame}{Limitations of my work}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{itemize}\setlength{\itemsep}{0.6em}
        \item \textbf{Noise models, not hardware.} ESP and TVD come from calibration snapshots, with gate errors only (no readout, idle or crosstalk). Ion parameters are assumptions.
        \item \textbf{Global phase is not free for us.} Strict mode costs 240 single-qubit gates over the Heron suite. With phase as metadata (as Qiskit, TKET and BQSKit do), all gates \(\times\)0.838 \(\to\) \(\times\)0.825.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \two{
      \begin{itemize}\setlength{\itemsep}{0.6em}
        \item \textbf{Baselines.} BQSKit (approximate) needs fewer 2q gates than we do. Quartz/QUESO were not run, and Qiskit, TKET and BQSKit use one seed.
        \item \textbf{Local only.} Layout and routing are Qiskit's, windows have \(\le3\) qubits, and the code is single-threaded Python (median 10\,s, max 180\,s per circuit vs.\ \(<\)1\,s for Qiskit).
        \item \textbf{Scope.} Circuits up to \(\sim\)1,000 gates, and no mid-circuit measurement or feed-forward.
      \end{itemize}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Weaknesses of the original paper}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{enumerate}\setlength{\itemsep}{0.4em}\small
        \item \textbf{Missing structure.} Clifford gate sets, but no tableaux and no exact keys, floating-point lookup with three tolerances, and a classifier to speed up an avoidable \(O(N)\) scan.
        \item \textbf{No notion of optimality.} Only compared with Qiskit, and 4.5\(\times\) from an optimum computable in minutes.
        \item \textbf{Equivalence.} Up to phase only (unsound under control), with no state/measurement context and no error tracking.
        \item \textbf{Hardware model.} Discrete angles, all-to-all, total count. Matches neither IBM nor ion hardware.
      \end{enumerate}
    \end{column}
    \begin{column}{0.48\textwidth}
      \two{
      \begin{enumerate}\setcounter{enumi}{4}\setlength{\itemsep}{0.4em}\small
        \item \textbf{Evaluation.} Random benchmarks mix, so ``from 300 gates'' measures input length. One circuit per table, and MATLAB vs.\ compiled timings.
        \item \textbf{Search.} Contiguous windows, random swaps and no convergence criterion. Radius 5 is the binding limit.
        \item \textbf{V3.} Unreported features and threshold, 30\,\% recall at default.
        \item \textbf{Hardware evidence.} One 2-qubit circuit and no error bars. The effect vanishes at Qiskit level 3.
        \item \textbf{Reproducibility.} Only V2 released, with visible bugs, and the database for Tables III--V unstated.
      \end{enumerate}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Summary}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{About the paper}
        \begin{itemize}\setlength{\itemsep}{0.3em}
          \item Reproduces, with qualifications on database, forest threshold, baselines and transpiler level.
          \item Its gates are Clifford, and exact optima show it is 3--4.5\(\times\) from optimal.
          \item Theory explains why. Radius 5 and window 8 sit below the exactness threshold, and random benchmarks measure a constant.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \two{
      \begin{exampleblock}{About the extension}
        \begin{itemize}\setlength{\itemsep}{0.3em}
          \item A general optimiser built on ``equivalence = group action'', with soundness and error bounds.
          \item Proven optimum on 35/40 of the paper's 3-qubit circuits without Clifford knowledge.
          \item On 58 real algorithms, stacked on Qiskit L3, it never adds 2q gates, never lowers ESP and removes 13\,\% of gates on Heron.
          \item BQSKit's approximate resynthesis needs fewer 2q gates. Exact rewriting keeps the unitary, removes more single-qubit gates and is faster. Run after BQSKit, it removes a further 16\,\% of gates on Heron.
        \end{itemize}
      \end{exampleblock}}
    \end{column}
  \end{columns}
\end{frame}
